#!BAT!
close all
cls
test init

!__JS__!
$TEST_DATA = [
  {src: '', key: '', b64: ''},
  {src: '', key: 'x', b64: ''},
  {src: 'abc', key: '', b64: 'YWJj'},
  {src: 'abc', key: 'x', b64: 'GRob'},
  {src: 'abc', key: 'xyz', b64: 'GRsZ'},
  {src: 'abc', key: 'xyz1', b64: 'GRsZ'},
  {src: 'a', key: 'A2345678901234567890123456789012345678901234567890123456789012345678901234567890123456789012345678901234567890123456789012345678901234567890123456789012345678901234567890123456789012345678901234567890123456789012345678901234567890123456789012345678901234#', b64: 'IA=='},
  {src: 'あいう', key: '', b64: '44GC44GE44GG'},
  {src: 'あいう', key: 'x', b64: 'm/n6m/n8m/n+'},
  {src: 'あいう', key: 'xyz', b64: 'm/j4m/j+m/j8'},
  {src: 'あいう', key: 'xyz123456a', b64: 'm/j40rO317Sw'}
]
!__JS__!

#----------------------------------------------------------
test set id "xb64"

${i} = 0
LOOP (${i} < $TEST_DATA.length) (
  ${ret} = xb64.encode($TEST_DATA[${i}].src, $TEST_DATA[${i}].key)
  test verify ${ret} == $TEST_DATA[${i}].b64
  ${i}++
)

${i} = 0
LOOP (${i} < $TEST_DATA.length) (
  ${ret} = xb64.decodeToString($TEST_DATA[${i}].b64, $TEST_DATA[${i}].key)
  test verify ${ret} == $TEST_DATA[${i}].src
  ${i}++
)

test end
test result
